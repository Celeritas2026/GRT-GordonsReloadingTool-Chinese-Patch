======插件清单======插件清单是一个 XML 格式的文件，文件名固定（全部为小写字母）：==com.grt.plugin.xml==一个常见的插件清单具有如下结构：<file com.grt.plugin.xml><?xml version="1.0" encoding="UTF-8" standalone="yes" ?><GordonsReloadingTool><!--GRT 插件清单插件配置 <plugin>属性 "id" 必须以域名 "com.grt.plugin" 开头，通常在其后附加一个不含空格的、唯一的字母数字名称作为您的插件名。属性 "enabled" 用于启用或停用插件。若为 "false"，启动时将忽略该插件。属性 "launch-windows" 和 "launch-linux" 定义插件的可执行文件。可执行文件也可以放在子文件夹中。此时只需附上本地目录路径，例如 Windows 下为 "bin\plugin.exe"，Linux 下为 "bin/plugin"。这两个属性中至少要有 ONE 个包含有效的可执行文件。若未指定对应操作系统的文件，插件将无法加载。属性 "launch-type" 配置插件的启动方式。值为 "permanent" 时，插件在启动时执行并持续挂接到 GRT（默认）。值为 "onDemand" 时，仅当发生用户操作且插件尚未运行时才启动插件。插件可在完成一次处理后自行退出，并在下次用户操作时再次启动。如果您使用的是可在 Windows 和 Linux 上运行的脚本语言（例如 python），则在两个属性中填入同一个文件即可。属性 "launch-args" 是插件的自定义命令行，其中可使用占位符"{PLUGIN_COMMANDLINE}" 来代入来自 GRT 的原始命令行（其中包含特定的连接信息）。"launch_args" 会替换插件启动时传入的原始命令行。-->  <plugin	  name           = "GRT Example Plugin"	  id             = "com.grt.plugin.example"	  version        = "1.0.0"	  provider       = "Test Provider"    description    = "This is a plugin example"    launch-windows = "plugin.exe"    launch-linux   = "plugin"    launch-type    = "permanent"    launch-args    = ""    start-timeout  = "2.0"    enabled        = "false"  ><!--注册 GRT 事件每个以 "true" 启用的事件，在事件发生时都会发送给您的插件。注意：只启用真正需要的事件，否则在多个插件同时活动的情况下可能会影响 GRT 的性能。此处的事件仅作示例用途而全部启用。-->    <registerevent>      <event name="ColorPresetChange" enabled="true" />      <event name="TabSwitch" enabled="true" />      <event name="TabDataChange" enabled="true" />      <event name="TabUnitChange" enabled="true" />      <event name="TabClosed" enabled="true" />      <event name="WindowActivate" enabled="true" />      <event name="WindowDeactivate" enabled="true" />      <event name="WindowMaximize" enabled="true" />      <event name="WindowMinimize" enabled="true" />      <event name="WindowRestore" enabled="true" />      <event name="WindowMoved" enabled="true" />      <event name="WindowResized" enabled="true" />      <event name="WindowClosed" enabled="true" />    </registerevent><!--菜单集成您可以按需定义任意数量的菜单项和子菜单。主标签 <menu> 仅仅是以上面定义的插件名称创建的主菜单项。<menu> 只有 "autoenable" 和 "icon" 两个属性。若 autoenable="true"，GRT 会自动为您启用 GRT 菜单集成中的菜单项；否则您必须自行设置每个菜单项的启用/停用状态（使用 menuitem 命令）。每个 <menuitem> 都可以通过插入更多 <menuitem> 标签而成为子菜单。每个 <menuitem> 都必须具有 "id" 和 "label" 属性。属性 "id" 会附加到上面定义的插件 ID 之后，并在发生 menuaction事件时发送给插件。（下方的工具栏图标/按钮同理）-->    <menu autoenable="true" icon="media/icons/icn_about_16x16.png">      <menuitem id="menu1" label="Menu 1">    		<menuitem id="entry1" label="Entry 1" icon="media/icons/icn_plugin_16x16.png" enabled="true" checked="false"/>    		<separator />    		<menuitem id="entry2" label="Entry 2" icon="media/icons/icn_plugin_go_16x16.png" enabled="true" checked="false"/>    		<menuitem id="entry3" label="Entry 3" icon="media/icons/icn_plugin_edit_16x16.png" enabled="true" checked="false"/>    	</menuitem>      <menuitem id="menu2" label="Menu 2">    		<menuitem id="entry1" label="Entry 1" icon="media/icons/icn_plugin_16x16.png" enabled="false" checked="false"/>    		<menuitem id="entry2" label="Entry 2" enabled="true" checked="true"/>    	</menuitem>  		<menuitem id="about" label="About" icon="media/icons/icn_about_16x16.png" enabled="true" checked="false"/>  		<menuitem id="version" label="Version" enabled="true" checked="false"/>    </menu><!--工具栏集成您可以按需定义任意数量的工具栏图标/按钮。也可以用 <separator /> 添加分隔符。每个工具栏图标必须为 32 x 32 像素，GRT 会在其右下角叠加一个小的插件图标，以标示该工具栏图标/按钮来自插件。-->    <toolbar>      <toolbaritem        id           = "toolbar_button1"        label        = "Plugin example"        tooltip      = "press to show this plugin"        icon         = "media/icons/icn_about_32x32.png"        enabled      = "true"      />    </toolbar>  </plugin></GordonsReloadingTool></file>
